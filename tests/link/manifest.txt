# Link-test manifest, run by `make linktest' (tools/linktest.py run).
#
# One unit per line, as in tests/conformance/manifest.txt:
#   FILE SPEC... [ALIGN] [A5]...
# The unit is compiled as the conformance harness compiles it; the C
# objects of ../source that define its globals are dropped from a scratch
# copy of the kernel build and the dpc object is linked in their place
# (tests/link/README.md, which also lists the units that cannot link yet
# and why).

# The module-local 0xE00EB0 is a local symbol of the unit; ../source's
# ZERO_PAGE (misc/zero_page.c) stays for its other C callers.  The unit
# names AST_$ZERO_BUFF, an absolute map address (absdefs.o).
tests/conformance/ast_page_zero.pas          0xE00EB0 AST_$PAGE_ZERO

# A `bsr.w' to a routine of the module outside the unit (module-routines=):

tests/conformance/pchist_interrupt.pas       PCHIST_$INTERRUPT  module-routines=PCHIST_$COUNT

# Module data (bead domain-pascal-3bp): the `--section-addr' of a data
# block gives its image address; the harness finds the C object of
# ../source that starts there (a MODULE_DATA_DEFINE block, a C object
# with a map name, or a DATA_ANCHORS entry of gen_layout_ld.py) and
# compiles the unit with `dpc --data-base=SECTION=SYMBOL': no block of
# its own, `lea SYMBOL.l,A5'.
tests/conformance/proc1_get_loadav.pas       PROC1_$GET_LOADAV  --section-addr .bss.moddata.PROC1_.WIRED_DATA=0xE254E8
tests/conformance/proc2_get_boot_flags.pas   PROC2_$GET_BOOT_FLAGS  --section-addr .bss.moddata.PROC2=0xE7BE84
tests/conformance/fim_get_fim_addr.pas       FIM_$GET_FIM_ADDR  --section-addr .bss.moddata.FIM_=0xE2126C
tests/conformance/smd_busy_wait.pas          SMD_$PUTC SMD_$BUSY_WAIT SMD_$LITES  --section-addr .bss.moddata.SMD.WIRED_DATA=0xE2E060  a5-always-from=SMD_$BUSY_WAIT
tests/conformance/mmap_alloc_contig.pas      MMAP_$ALLOC_CONTIG  --section-addr .bss.moddata.MMAP_=0xE23284  a5-always=MMAP_$ALLOC_CONTIG
tests/conformance/pkt_next_id.pas            PKT_$NEXT_ID  --section-addr .bss.moddata.PKT=0xE24C9C
